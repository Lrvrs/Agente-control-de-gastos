# Dependencias de la aplicacion. Version fijada en todas ellas a proposito:
# una actualizacion inesperada el dia de la clase es un riesgo real y evitable.
#
# IMPORTANTE: este proyecto requiere Python 3.11, que se selecciona en el panel
# de Streamlit Community Cloud (Settings > Python version). No existe forma de
# fijarlo desde el repositorio: Streamlit no lee runtime.txt. Si se despliega
# con una version mas reciente de Python, Pillow -que llega como dependencia
# de Streamlit- no encuentra paquete precompilado, intenta compilarse desde el
# codigo fuente y falla por ausencia de las cabeceras de zlib en la maquina de
# construccion.
streamlit==1.39.0

# openai 1.54.x pasaba el argumento `proxies` al construir su cliente HTTP.
# httpx elimino ese argumento en la version 0.28, de modo que la combinacion
# openai 1.54 con httpx 0.28 falla con un TypeError en cuanto se instancia el
# cliente. Se fija una version de openai posterior a esa correccion y se ancla
# httpx de forma explicita para que una resolucion futura no vuelva a emparejar
# dos versiones incompatibles.
openai==1.109.1
httpx==0.28.1

# Lectura de los ficheros .xlsx que suben los alumnos. Es Python puro, no
# requiere compilacion y por tanto no reintroduce el fallo de construccion
# que provocaron pillow y pandas en el primer despliegue.
openpyxl==3.1.5
